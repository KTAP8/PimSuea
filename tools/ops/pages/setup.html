R2_ACCESS_KEY_ID=...
R2_SECRET_ACCESS_KEY=...
R2_PUBLIC_URL_ASSETS=https://...
R2_PUBLIC_URL_PRINT=https://...
R2_PUBLIC_URL_PRINT_ORDERED=https://...
